import json, os

def process(path, threshold=10, out=None):
    data = json.load(open(path))
    result = []
    for i in range(len(data)):
        item = data[i]
        if item.get("active") == True:
            score = 0
            for k in item["metrics"]:
                score = score + item["metrics"][k]
            if score > threshold:
                name = item["name"].strip().lower()
                tags = []
                for t in item.get("tags", []):
                    if t not in tags:
                        tags.append(t)
                result.append({"name": name, "score": score, "tags": tags})
    result.sort(key=lambda r: r["score"])
    if out != None:
        f = open(out, "w")
        json.dump(result, f, indent=2)
    total = 0
    for r in result:
        total += r["score"]
    avg = total / len(result)
    print("processed", len(result), "items, avg", avg)
    return result
